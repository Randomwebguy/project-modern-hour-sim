# Compile every shader entry point at the batch boundary, without running the game.
find_program(GENZH_FXC fxc HINTS
    "$ENV{WindowsSdkDir}/bin/$ENV{WindowsSDKVersion}/x86"
    "$ENV{WindowsSdkDir}/bin/$ENV{WindowsSDKVersion}/x64"
)
if(NOT GENZH_FXC)
    message(FATAL_ERROR "fxc from the Windows SDK is required for shader validation")
endif()
set(GENZH_SHADER_OBJECTS)
foreach(entry IN ITEMS VSMain PSMain PSShadow VSPresent PSTonemap PSSceneAA)
    if(entry MATCHES "^VS")
        set(profile vs_5_0)
    else()
        set(profile ps_5_0)
    endif()
    set(object "${CMAKE_CURRENT_BINARY_DIR}/${entry}.cso")
    add_custom_command(OUTPUT "${object}"
        COMMAND "${GENZH_FXC}" /nologo /WX /Ges /O3 /T ${profile} /E ${entry}
            /Fo "${object}" "${CMAKE_CURRENT_SOURCE_DIR}/Scene.hlsl"
        DEPENDS Scene.hlsl MaterialLighting.hlsl DirectionalPCSS.hlsl VERBATIM
    )
    list(APPEND GENZH_SHADER_OBJECTS "${object}")
endforeach()
add_custom_target(genzh_d3d11_shaders DEPENDS ${GENZH_SHADER_OBJECTS})
add_dependencies(genzh_d3d11_renderer genzh_d3d11_shaders)
